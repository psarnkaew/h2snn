"""Historical DCV candidate diagnostic retained for provenance only.

Dump the AMPSS candidate pool with its scores, before and after the candidate cap.

Answers the question `problem/ampss_issues.md` §3 asks: for a given config, is a meta-path
absent from the selected pool because it was never *generated* (cut by `max_schema_candidates`,
which truncates the DFS in pre-order) or because it was generated and *lost on scoring*?

    conda run -n h2snn python scripts/diagnose_ampss_candidates.py \
        --config configs/experiments/acm_h2snn_ampss.yaml

Everything runs on CPU and materialises no faces — only Stages B–D of schema discovery.
"""

from __future__ import annotations

import argparse
import sys
from collections import Counter
from pathlib import Path

import torch
import yaml

PROJECT_ROOT = Path(__file__).resolve().parent.parent
if str(PROJECT_ROOT) not in sys.path:
    sys.path.insert(0, str(PROJECT_ROOT))

from src.configs.config import TrainConfig
from src.data.dataset import _load_hetero_data, _prepare_target_labels
from src.data.hetero_flatten import flatten_edges, stack_node_features
from src.data.preprocessing import label_propagation
from src.data.schema_discovery import (
    _MAX_STATS_SAMPLE,
    estimate_schema_stats,
    generate_schema_candidates,
    jaccard_endpoint_overlap,
    select_schema_pool,
    topology_aware_score,
)
from src.data.schema_structures import SchemaCandidate, SchemaStats

_UNCAPPED = 10**9


def format_schema(candidate: SchemaCandidate) -> str:
    """Relation-typed rendering: `paper -[to]-> author -[to]-> paper`."""
    parts = [candidate.node_types[0]]
    for relation_name, node_type in zip(candidate.relation_names, candidate.node_types[1:]):
        parts.append(f"-[{relation_name}]-> {node_type}")
    return " ".join(parts)


def _generate(cfg: TrainConfig, relation_tuples, max_candidates: int):
    discovery = cfg.discovery
    return generate_schema_candidates(
        relation_tuples=relation_tuples,
        target_node_type=cfg.data.target_node_type,
        min_length=discovery.min_schema_length,
        max_length=discovery.max_schema_length,
        require_target_start=discovery.require_target_start,
        require_target_end=discovery.require_target_end,
        allow_repeated_node_types=discovery.allow_repeated_node_types,
        max_identities=max_candidates,
    )


def _report_pool(
    title: str,
    candidates: tuple[SchemaCandidate, ...],
    stats: tuple[SchemaStats, ...],
    selected_schema_k: int,
    top_n: int,
) -> None:
    print(f"\n=== {title}: {len(candidates)} candidates ===")
    ranked = sorted(
        zip(candidates, stats),
        key=lambda pair: -topology_aware_score(pair[1], []),
    )
    print(f"  top {min(top_n, len(ranked))} by q0 (score against an empty selection):")
    for rank, (candidate, stat) in enumerate(ranked[:top_n]):
        print(
            f"    {rank:3d} q0={topology_aware_score(stat, []):.6f} "
            f"D={stat.endpoint_label_disc:.6f} C={stat.end_coverage:.4f} "
            f"V={stat.type_diversity:.4f} n={stat.num_instances:6d} | "
            f"{format_schema(candidate)}"
        )

    selected, selected_stats, topology_scores = select_schema_pool(
        candidates, stats, selected_schema_k
    )
    print(f"\n  greedy selection (k={selected_schema_k}):")
    for i, (candidate, stat) in enumerate(zip(selected, selected_stats)):
        print(
            f"    {i}: {format_schema(candidate)}  "
            f"D={stat.endpoint_label_disc:.6f} C={stat.end_coverage:.4f} "
            f"final_q={topology_scores[i]:.6f}"
        )

    print("\n  node-type patterns among the selected:")
    patterns = Counter("-".join(c.node_types) for c in selected)
    for pattern, count in patterns.most_common():
        print(f"    {count}x  {pattern}")
    print(f"    -> {len(patterns)} distinct pattern(s) across {len(selected)} slots")

    print("\n  pairwise endpoint Jaccard among the selected (the §2 redundancy signal):")
    for i in range(len(selected_stats)):
        row = " ".join(
            f"{jaccard_endpoint_overlap(selected_stats[i], selected_stats[j]):.2f}"
            for j in range(len(selected_stats))
        )
        print(f"    {row}")


def main(cfg: TrainConfig, top_n: int) -> None:
    data = _load_hetero_data(cfg)
    raw_x, _, _, node_slices, _ = stack_node_features(data)
    edge_index, edge_type, _, relation_tuples = flatten_edges(data, node_slices)

    print("=== relation table (id: src -[name]-> dst) ===")
    for relation_id, (src, name, dst) in enumerate(relation_tuples):
        print(f"  {relation_id}: {src} -[{name}]-> {dst}")

    (
        _,
        _,
        _,
        _,
        target_indices,
        target_y,
        target_train_mask,
        num_classes,
        _,
    ) = _prepare_target_labels(
        data=data,
        node_slices=node_slices,
        target_node_type=cfg.data.target_node_type,
        val_ratio=cfg.data.acm_val_ratio,
        split_seed=cfg.data.acm_split_seed,
    )
    propagated_labels = label_propagation(
        edge_index=edge_index,
        train_indices=target_indices[target_train_mask],
        train_labels=target_y[target_train_mask],
        num_nodes=raw_x.size(0),
        num_classes=num_classes,
        num_steps=cfg.data.num_label_prop_steps,
    )

    cap = cfg.discovery.max_schema_candidates
    capped = _generate(cfg, relation_tuples, cap)
    uncapped = _generate(cfg, relation_tuples, _UNCAPPED)

    print(f"\n=== candidate generation ===")
    print(f"  max_schema_candidates={cap} -> {len(capped)} candidates")
    print(f"  uncapped                 -> {len(uncapped)} candidates")
    if len(uncapped) > len(capped):
        print(f"  *** {len(uncapped) - len(capped)} candidate(s) CUT by the cap ***")
        print("  cut candidates (DFS pre-order index in the uncapped list):")
        capped_signatures = {(c.node_types, c.relation_ids) for c in capped}
        for index, candidate in enumerate(uncapped):
            if (candidate.node_types, candidate.relation_ids) not in capped_signatures:
                print(f"    [{index:4d}] {format_schema(candidate)}")

    stats_args = dict(
        node_slices=node_slices,
        edge_index=edge_index,
        edge_type=edge_type,
        propagated_labels=propagated_labels,
        target_node_type=cfg.data.target_node_type,
        max_instances_per_anchor=cfg.discovery.max_instances_per_anchor,
        split_seed=cfg.data.acm_split_seed,
    )
    capped_stats = estimate_schema_stats(candidates=capped, **stats_args)
    _report_pool(
        f"CAPPED pool (max_schema_candidates={cap}) — what actually runs",
        capped,
        capped_stats,
        cfg.discovery.selected_schema_k,
        top_n,
    )

    if len(uncapped) > len(capped):
        uncapped_stats = estimate_schema_stats(candidates=uncapped, **stats_args)
        _report_pool(
            "UNCAPPED pool — what the cap is hiding",
            uncapped,
            uncapped_stats,
            cfg.discovery.selected_schema_k,
            top_n,
        )

    # §3b: how much of the graph the stats are actually estimated from.
    target_begin, target_end = node_slices[cfg.data.target_node_type]
    num_targets = target_end - target_begin
    max_anchors = _MAX_STATS_SAMPLE // max(cfg.discovery.max_instances_per_anchor, 1) + 1
    saturated = sum(1 for s in capped_stats if s.num_instances >= _MAX_STATS_SAMPLE)
    print("\n=== §3b: stats sampling coverage ===")
    print(f"  target '{cfg.data.target_node_type}' nodes: {num_targets}")
    print(
        f"  _MAX_STATS_SAMPLE={_MAX_STATS_SAMPLE}, "
        f"max_instances_per_anchor={cfg.discovery.max_instances_per_anchor}"
    )
    print(
        f"  -> at most ~{max_anchors} anchors visited "
        f"({max_anchors / max(num_targets, 1):.1%} of targets), always the lowest node ids"
    )
    print(f"  num_instances saturated at the cap for {saturated}/{len(capped_stats)} candidates")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--config", type=Path, required=True)
    parser.add_argument("--top_n", type=int, default=15)
    args = parser.parse_args()

    payload = yaml.safe_load(args.config.read_text(encoding="utf-8"))
    payload["device"] = "cpu"
    torch.set_grad_enabled(False)

    main(TrainConfig.from_dict(payload), args.top_n)
